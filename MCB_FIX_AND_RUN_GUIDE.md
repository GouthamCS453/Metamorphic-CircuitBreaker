### 3.4 Scoring functions (replace your IR/CBI/state code with this)

```python
import numpy as np, json

W = {"mild": 1.0, "moderate": 0.6, "severe": 0.3, "single": 1.0}
VARIANTS = json.load(open("variants.json"))
FAMS = ["geometric", "photometric", "sensor_noise"]

def type_scores(flips):
    """flips: (N, n) 0/1 -> dict[(family,type)] = Score_m, shape (N,).  Eq. 2."""
    groups = {}
    for j, v in enumerate(VARIANTS):
        groups.setdefault((v["family"], v["type"]), {}).setdefault(v["tier"], []).append(j)
    out = {}
    for key, tiers in groups.items():
        num = sum(W[t] * flips[:, idx].mean(1) for t, idx in tiers.items())  # fraction of variants in tier that flipped
        out[key] = num / sum(W[t] for t in tiers)
    return out

def family_ir(flips):
    """Eq. 3: unweighted mean over exactly 3 types per family. Returns dict[family] -> (N,)."""
    s = type_scores(flips)
    ir = {}
    for f in FAMS:
        ms = [v for (fam, _), v in s.items() if fam == f]
        assert len(ms) == 3, f"family {f} has {len(ms)} transform types, need 3"
        ir[f] = np.mean(ms, axis=0)
    return ir, s

def cbi(ir, c0, tau_fam, a=0.50, b=0.35, g=0.15):
    """Eq. 4 and 5."""
    M = np.stack([ir[f] for f in FAMS], 1)
    cfs = (M >= tau_fam).mean(1)
    return a * M.max(1) + b * cfs + g * (1 - c0)

def cbi_flat(flips, c0, a=0.50, b=0.35, g=0.15):
    """Ablation A1: flat weighted flip score + same gamma term."""
    w = np.array([W[v["tier"]] for v in VARIANTS])
    flat = (flips * w).sum(1) / w.sum()
    return (a + b) * flat + g * (1 - c0)

def state(cbi_val, theta_trip):
    theta_warn = round(0.5 * theta_trip, 4)
    return np.where(cbi_val >= theta_trip, "OPEN",
           np.where(cbi_val >= theta_warn, "HALF_OPEN", "CLOSED"))
```

`alpha, beta, gamma = 0.50, 0.35, 0.15` in every domain. Only `tau_fam` and `theta_trip` are calibrated. Make `CircuitBreakerConfig` set `theta_warn = 0.5 * theta_trip` automatically and assert `theta_warn < theta_trip`.

@torch.no_grad()
def mcdo_score(feat, head, p=0.2, T=10):
    probs = torch.stack([torch.softmax(head(torch.nn.functional.dropout(feat, p, training=True)), -1) for _ in range(T)]).mean(0)

## 6. Calibration on Partition A (patched Test 5) and evaluation on Partition B

### 6.1 Metrics helpers

```python
import numpy as np
from sklearn.metrics import roc_auc_score, average_precision_score

def wilson(k, n, z=1.96):
    if n == 0: return (None, None)
    p = k / n; d = 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * np.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (max(0.0, c - h), min(1.0, c + h))

def confusion(err, trip):
    tp = int((err & trip).sum()); fn = int((err & ~trip).sum())
    fp = int((~err & trip).sum()); tn = int((~err & ~trip).sum())
    return tp, fp, tn, fn

def rates(err, trip):
    tp, fp, tn, fn = confusion(err, trip)
    rec = tp / (tp + fn) if tp + fn else np.nan; far = fp / (fp + tn) if fp + tn else np.nan
    return dict(tp=tp, fp=fp, tn=tn, fn=fn, recall=rec, far=far, youden=rec - far,
                recall_ci=wilson(tp, tp + fn), far_ci=wilson(fp, fp + tn),
                precision=tp / (tp + fp) if tp + fp else np.nan)

def pick_threshold(score, err, rule, grid=None):
    """rule 'op_j': max Youden J (tie -> lower FAR). rule 'op_far10': highest recall with FAR<=0.10."""
    cands = np.unique(score) if grid is None else grid
    best, best_key = None, None
    for t in cands:
        r = rates(err, score >= t)
        if rule == "op_j": key = (r["youden"], -r["far"])
        else:
            if r["far"] > 0.10: continue
            key = (r["recall"], -r["far"])
        if best_key is None or key > best_key: best, best_key = t, key
    return best
```

### 6.4 Paired bootstrap for comparisons (group-aware)

```python
def paired_auc_diff(err, s1, s2, groups, B=2000, seed=0):
    rng = np.random.default_rng(seed); ug = np.unique(groups)
    idx_by_g = {g: np.where(groups == g)[0] for g in ug}; diffs = []
    for _ in range(B):
        idx = np.concatenate([idx_by_g[g] for g in rng.choice(ug, len(ug))])
        if err[idx].min() == err[idx].max(): continue
        diffs.append(roc_auc_score(err[idx], s1[idx]) - roc_auc_score(err[idx], s2[idx]))
    return float(np.mean(diffs)), tuple(np.percentile(diffs, [2.5, 97.5]))
```

### 8.1 Design

- Each image gets **one family's corruption**, using corruptions **not in the bank**:

```python
import numpy as np, cv2

def motion_blur(img, k=11, ang=0):
    ker = np.zeros((k, k)); ker[k // 2, :] = 1; M = cv2.getRotationMatrix2D((k / 2 - .5, k / 2 - .5), ang, 1)
    ker = cv2.warpAffine(ker, M, (k, k)); ker /= ker.sum(); return cv2.filter2D(img, -1, ker)
def defocus_blur(img, r=3):
    ker = np.zeros((2 * r + 1, 2 * r + 1)); cv2.circle(ker, (r, r), r, 1, -1); ker /= ker.sum(); return cv2.filter2D(img, -1, ker)
def shot_noise(img, scale=40, rng=None):                       # img uint8
    rng = rng or np.random.default_rng(); return np.clip(rng.poisson(img / 255 * scale) / scale * 255, 0, 255).astype(np.uint8)
def salt_pepper(img, p=0.02, rng=None):
    rng = rng or np.random.default_rng(); m = rng.random(img.shape[:2]); o = img.copy(); o[m < p / 2] = 0; o[m > 1 - p / 2] = 255; return o
def gamma(img, g=2.0): return np.clip(255 * (img / 255.0) ** g, 0, 255).astype(np.uint8)
def vignette(img, s=0.6):
    h, w = img.shape[:2]; y, x = np.ogrid[:h, :w]; d = np.sqrt(((x - w / 2) / (w / 2)) ** 2 + ((y - h / 2) / (h / 2)) ** 2)
    return np.clip(img * (1 - s * np.clip(d, 0, 1) ** 2)[..., None], 0, 255).astype(np.uint8)
def white_balance(img, r=1.10, b=0.90): o = img.astype(float); o[..., 0] *= r; o[..., 2] *= b; return np.clip(o, 0, 255).astype(np.uint8)
def translate(img, f=0.10):
    h, w = img.shape[:2]; M = np.float32([[1, 0, f * w], [0, 1, f * h]]); return cv2.warpAffine(img, M, (w, h), borderMode=cv2.BORDER_REFLECT)
def perspective(img, f=0.06):
    h, w = img.shape[:2]; s = np.float32([[0, 0], [w, 0], [w, h], [0, h]]); d = s + np.float32([[f * w, f * h], [-f * w, f * h], [-f * w, -f * h], [f * w, -f * h]]) * 0.5
    return cv2.warpPerspective(img, cv2.getPerspectiveTransform(s, d), (w, h), borderMode=cv2.BORDER_REFLECT)
def crop_resize(img, f=0.10):
    h, w = img.shape[:2]; c = img[int(f * h):h - int(f * h), int(f * w):w - int(f * w)]; return cv2.resize(c, (w, h))
```

### 8.2 Attribution rule

Normalize each family by the **clean correct samples of Partition A**: `z_f = (IR_f − mean_f) / std_f` (mean and std from Partition A, correct predictions only; use a small epsilon for std). Predicted family = `argmax_f z_f`; if all `IR_f == 0` the sample is **unattributable**. Also compute the raw `argmax IR_f` result.

### 8.3 Report

- Trip rate (frozen `mcb_op_j` thresholds) per injected family and for the clean control.
- 3×3 confusion matrix plus an "unattributable" column.
- Attribution accuracy overall and per family, over **all injected samples** (primary) and over tripped samples only (secondary), each with a Wilson 95% CI, and a binomial test against chance (1/3).
- Family-specific lift: mean `IR_f` injected vs clean control.
- Cross-contamination: fraction where a wrong family's `IR` exceeds `tau_fam*`.
- Decision rule **D4**.
